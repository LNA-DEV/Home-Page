{{- /* Content adapter: one page per dex species, per language. data/dex.yaml holds ~190 species and the site is trilingual, so hand-written page bundles would mean ~570 markdown files. The adapter builds them at build time instead, which keeps the species data in one place and still gives every species a real URL, a place in the search index and a sitemap entry. The page body is the species description in the current language (English fallback), so `.Plain` has something for the search index to chew on. The rendering itself is dispatched by params.theme = "dex-species", the same way the other gallery views work. */ -}} {{- $lang := .Site.Language.Lang -}} {{- $index := partialCached "dex-index.html" . -}} {{- range $index.all -}} {{- $species := .species -}} {{- $name := partial "dex-text.html" (dict "field" $species.names "lang" $lang) -}} {{- $name = $name | default $species.scientific | default .slug -}} {{- $description := partial "dex-text.html" (dict "field" $species.description "lang" $lang) -}} {{- /* Which language actually supplied the description, mirroring the fallback in dex-text.html, so the CC BY-SA credit on the page can link the exact Wikipedia article the text was abridged from. Every language's text comes from its own Wikipedia, so this is usually just $lang — but the handful of species with no article in a language fall back to the English text, and must then credit the English article rather than a page that did not supply the words. Empty where the article could not be resolved at all, and the credit then renders unlinked. */ -}} {{- $sourceLang := "" -}} {{- with $species.description -}} {{- if index . $lang -}} {{- $sourceLang = $lang -}} {{- else if index . "en" -}} {{- $sourceLang = "en" -}} {{- end -}} {{- end -}} {{- $sourceURL := "" -}} {{- if $sourceLang -}} {{- $sourceURL = index ($species.description_source | default dict) $sourceLang | default "" -}} {{- end -}} {{- /* dex_scientific / dex_family / dex_wikidata carry the taxon facts for the JSON-LD in schema_json.html, so the does not have to reach back into dex-index.html for them. */ -}} {{- $page := dict "path" .slug "kind" "page" "title" $name "description" (printf "%s — %s" $name ($species.scientific | default "")) "content" (dict "mediaType" "text/markdown" "value" $description) "params" (dict "theme" "dex-species" "dex_slug" .slug "translationKey" (printf "dex-%s" .slug) "hideMeta" true "searchHidden" false "filter_species" .scientific "dex_scientific" .scientific "dex_family" ($species.family | default "") "dex_wikidata" ($species.wikidata_id | default "") "dex_source_url" $sourceURL "include_archive" true "sort_by" "DateTaken" "sort_order" "desc" ) -}} {{- $.AddPage $page -}} {{- end -}}

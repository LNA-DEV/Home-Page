{{- /* gallery-project-nav.html "Previous project" / "Next project" under every gallery project page, the same on all three layouts: grid, essay and journal. Neighbours are the LISTED projects by weight, the order of the projects index (.CurrentSection.RegularPages leaves out an album with build.list: never). No wrapping: the first has no previous, the last no next — with two projects, wrapping would offer the same one twice. A page outside the list (an unlisted album, a demo) offers the first listed project as its next. Param: the project page. Renders nothing when there is nowhere to go. */ -}} {{- $page := . -}} {{- $list := slice -}} {{- with $page.CurrentSection }}{{ $list = .RegularPages.ByWeight }}{{ end -}} {{- $idx := -1 -}} {{- range $i, $p := $list }}{{ if eq $p $page }}{{ $idx = $i }}{{ end }}{{ end -}} {{- $prev := "" -}} {{- $next := "" -}} {{- if ge $idx 0 -}} {{- if gt $idx 0 }}{{ $prev = index $list (sub $idx 1) }}{{ end -}} {{- if lt (add $idx 1) (len $list) }}{{ $next = index $list (add $idx 1) }}{{ end -}} {{- else -}} {{- with $list }}{{ $next = index . 0 }}{{ end -}} {{- end -}} {{- if or $prev $next }}
{{- /* A slice, not a dict: ranging over a map goes in key order, which would put "next" before "prev". */}} {{- range slice (dict "dir" "prev" "page" $prev) (dict "dir" "next" "page" $next) }} {{- $dir := .dir }} {{- with .page }} {{- $g := partial "get-gallery.html" . }} {{- with $g }} {{- end }} {{ i18n (cond (eq $dir "prev") "project_previous" "project_next") }} {{ .Title }} {{- with $g }}{{ T "photoCount" .imageCount }}{{ end }} {{- end }} {{- end }}
{{- end -}}

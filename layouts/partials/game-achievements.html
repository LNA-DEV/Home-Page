{{- /* game-achievements.html One copy's achievements: a summary, filter/sort controls, and one row each. Usage: partial "game-achievements.html" (dict "doc" -.json> "store" "Steam" — the platform's display name "heading" true|false — name the store in the heading, when a game has achievement lists on two stores "lang" $lang) Default order: unlocked first, newest first; then locked, most common first — what you did, then what you could do next. The controls (filter all / unlocked / locked, sort default / rarity / A–Z) are client-side, assets/js/game-achievements.js, and rendered `hidden` so a visitor without JavaScript is not handed buttons that do nothing. A locked icon is the unlocked icon greyed by CSS: one file per achievement, not two (docs/concepts/gaming-game-pages.md §3). Icons are committed under static/images/games/achievements/ and served from this site — no request leaves it, which is what keeps the page usable on the Tor build. Steam delivers no description for a hidden achievement, locked or unlocked; such a row says so instead of leaving a blank line. */ -}} {{- $doc := .doc -}} {{- $lang := .lang | default site.Language.Lang -}} {{- $store := .store -}} {{- $items := $doc.achievements | default slice -}} {{- $done := slice -}} {{- $open := slice -}} {{- range $i, $a := $items -}} {{- $ts := 0 -}} {{- with $a.unlocked }}{{ $ts = (time.AsTime .).Unix }}{{ end -}} {{- $row := dict "a" $a "i" $i "ts" $ts "pct" (float ($a.percent | default -1)) -}} {{- if $a.achieved }}{{ $done = $done | append $row }}{{ else }}{{ $open = $open | append $row }}{{ end -}} {{- end -}} {{- $rows := (sort $done "ts" "desc") | append (sort $open "pct" "desc") -}} {{- $total := len $items -}} {{- $unlocked := len $done -}} {{- $share := 0.0 -}} {{- if gt $total 0 }}{{ $share = mul (div (float $unlocked) (float $total)) 100 }}{{ end -}}

{{- i18n "game_achievements" -}} {{- if .heading }} · {{ $store }}{{ end -}}

{{ $unlocked }} / {{ $total }}
{{ lang.FormatNumber 0 $share }} %
    {{- range $n, $row := $rows }} {{- $a := $row.a }} {{- $name := partial "dex-text.html" (dict "field" $a.name "lang" $lang) | default $a.key }} {{- $desc := partial "dex-text.html" (dict "field" $a.description "lang" $lang) }}
  1. {{- /* A static file, not a Hugo resource. Every resources.Get file lands in one LRU cache partition capped by COUNT (Hugo's dynacache: about 4,000 entries; memory is irrelevant, HUGO_MEMORYLIMIT does not move it). The site already sends ~1,200 files through it; the ~3,100 icons pushed it over, evicted resources were re-created on their next use and published a second time, and Hugo warned "Duplicate target paths" for all 150 dex range files and a varying handful of icons. Measured: fine at 3,104 icons, broken at 3,145; fine again with all icons but the gallery mount emptied. The icons need no processing, so static/ is where they belong anyway. The fileExists check covers the few icons Steam no longer serves. */}} {{- with $a.icon }}{{ if fileExists (printf "static/%s" .) }} {{- end }}{{ end }}
    {{ $name }} {{- with $desc }} {{ . }} {{- else }}{{ if $a.hidden }} {{ i18n "game_achv_hidden" (dict "store" $store) }} {{- end }}{{ end }}
    {{- if $a.achieved }} {{- with $a.unlocked }} {{ time.Format ":date_medium" (time.AsTime .) }} {{- else }} {{ i18n "game_achv_unlocked" }} {{- end }} {{- else }} {{ i18n "game_achv_locked" }} {{- end }} {{- if ge $row.pct 0.0 }} {{ lang.FormatNumber 1 $row.pct }} % {{- end }}
    2. {{- end }}

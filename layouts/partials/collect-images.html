{{- /* collect-images.html Central partial for fetching and filtering gallery images from global resources. Metadata is read from data/gallery.yaml via the gallery-meta.html lookup map. "filter" (dict "category" "animals")) }} Filter keys (all optional): - category: filter by category (animals, landscape, city, plants, portrait, others) - section: filter by section (general, archive) — if omitted, excludes archive - portfolio: if true, only return portfolio images - project: filter by project slug (matches any slug in the entry's `project`, which may be a single string or a list) - species: filter by scientific name, e.g. "Vulpes vulpes" (case-insensitive) - id: filter by the UUID from data/gallery.yaml — returns at most one image. It is inlined rather than calling that partial because this one is NOT cached at its call sites: it runs once per dex species page, so three partial calls per photo would be the better part of a million per build. */ -}} {{- $lang := site.Language.Lang -}} {{- /* Licences, resolved for this language. Cached per language: the label, the terms and a site-relative url all differ between them. */ -}} {{- $licenseIndex := partialCached "license-index.html" . $lang -}} {{- /* Tags the camera or the export chain wrote into dc:subject that must never reach a reader (data/galleryTagIgnore.yaml). match like dex-index. */ -}} {{- if ne (lower ($meta.species | default "")) (lower .) -}} {{- $include = false -}} {{- end -}} {{- end -}} {{- with index $filter "id" -}} {{- if ne ($meta.id | default "") . -}} {{- $include = false -}} {{- end -}} {{- end -}} {{- if $include -}} {{- /* Extract EXIF metadata */ -}} {{- $title := "" -}} {{- $date := "" -}} {{- $dateTaken := "" -}} {{- $aperture := "" -}} {{- $shutterSpeed := "" -}} {{- $iso := "" -}} {{- $focalLength := "" -}} {{- $cameraModel := "" -}} {{- $lensModel := "" -}} {{- $keywords := "" -}} {{- $software := "" -}} {{- $tagChunks := slice -}} {{- with $image.Meta -}} {{- $date = .Date -}} {{- /* .Date is a time.Time here; year > 2000 mirrors the feed's valid-date convention */ -}} {{- if gt .Date.Year 2000 -}} {{- $dateTaken = .Date.Format "2006-01-02" -}} {{- end -}} {{- /* Nothing editorial is read out of the file here any more — the one-time import (docs/concepts/gallery-metadata-import.md) moved the licence, alt, title and the seven dc:description captions into data/gallery.yaml. What is left below is the technical data and the tags, which stay darktable-owned by design. */ -}} {{- with .Exif.subject -}} {{- $keywords = . -}} {{- end -}} {{- if not $keywords -}} {{- with .Exif.Subject -}} {{- $keywords = . -}} {{- end -}} {{- end -}} {{- if not $keywords -}} {{- with .Exif.Keywords -}} {{- $keywords = . -}} {{- end -}} {{- end -}} {{- if not $keywords -}} {{- with .XMP.Subject -}} {{- $keywords = delimit . ", " -}} {{- end -}} {{- end -}} {{- if not $keywords -}} {{- with $meta.tags -}} {{- $keywords = delimit . ", " -}} {{- end -}} {{- end -}} {{- /* .Tags is the UNION of every source, unlike $keywords above, which is first-match-wins because it is what the info popup displays. The similarity scorer wants the richest vocabulary available: 352 photos carry keywords only in EXIF/XMP and others only in the data file, so preferring one source would throw away signal. */ -}} {{- range $chunk := (slice .Exif.subject .Exif.Subject .Exif.Keywords .XMP.Subject) -}} {{- with $chunk -}} {{- if reflect.IsSlice . -}} {{- $tagChunks = $tagChunks | append (delimit . ",") -}} {{- else -}} {{- $tagChunks = $tagChunks | append (printf "%v" .) -}} {{- end -}} {{- end -}} {{- end -}} {{- with .Exif.FNumber -}} {{- $fStr := printf "%v" . -}} {{- if strings.Contains $fStr "/" -}} {{- $parts := split $fStr "/" -}} {{- $num := float (index $parts 0) -}} {{- $den := float (index $parts 1) -}} {{- $aperture = printf "f/%.1f" (div $num $den) -}} {{- else -}} {{- $aperture = printf "f/%.1f" (float .) -}} {{- end -}} {{- end -}} {{- with .Exif.ExposureTime -}} {{- $expStr := printf "%v" . -}} {{- if strings.Contains $expStr "/" -}} {{- $shutterSpeed = $expStr -}} {{- else -}} {{- $exp := float . -}} {{- if gt $exp 0 -}} {{- if lt $exp 1.0 -}} {{- $shutterSpeed = printf "1/%ds" (int (math.Round (div 1.0 $exp))) -}} {{- else -}} {{- $shutterSpeed = printf "%.1fs" $exp -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- with .Exif.ISO -}} {{- $iso = printf "ISO %v" . -}} {{- end -}} {{- with .Exif.FocalLength -}} {{- /* EXIF stores focal length as a rational, and it is not always reduced — 61 photos here carry "419/50", which printed straight out reads "419/50mm" instead of "8.4mm". Same handling as FNumber and ExposureTime above. */ -}} {{- $fl := printf "%v" . -}} {{- if strings.Contains $fl "/" -}} {{- $parts := split $fl "/" -}} {{- $den := float (index $parts 1) -}} {{- if gt $den 0.0 -}} {{- $focalLength = printf "%.1fmm" (div (float (index $parts 0)) $den) -}} {{- end -}} {{- else -}} {{- $focalLength = printf "%vmm" . -}} {{- end -}} {{- end -}} {{- with .Exif.Model -}} {{- $cameraModel = . -}} {{- end -}} {{- with .Exif.LensModel -}} {{- $lensModel = . -}} {{- end -}} {{- with .Exif.Software -}} {{- $raw := printf "%v" . | strings.TrimSpace -}} {{- /* Cameras write their own FIRMWARE version into the Software tag — 157 of the 633 photos here say just "Ver.1.2", which is not editing software and would be noise in the info panel. Editing software always names itself, so a value that is nothing but a version number is rejected. */ -}} {{- if and $raw (not (findRE `(?i)^ver\.?\s*[0-9.]+$` $raw)) -}} {{- $software = $raw -}} {{- end -}} {{- end -}} {{- end -}} {{- with $meta.tags -}} {{- $tagChunks = $tagChunks | append (delimit . ",") -}} {{- end -}} {{- $tags := slice -}} {{- range $chunk := $tagChunks -}} {{- range $t := split (replace $chunk ";" ",") "," -}} {{- $t = lower (trim $t " ") -}} {{- if and $t (not (in $tags $t)) -}} {{- $drop := index $tagIgnoreSet $t -}} {{- if not $drop -}} {{- range $pat := $tagIgnorePatterns -}} {{- if findRE $pat $t -}}{{- $drop = true -}}{{- end -}} {{- end -}} {{- end -}} {{- if not $drop -}} {{- $tags = $tags | append $t -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Same blacklist over $keywords, which keeps its original casing and its first-match-wins source because it is the string the info popup prints. */ -}} {{- with $keywords -}} {{- $kept := slice -}} {{- range $t := split (replace . ";" ",") "," -}} {{- $t = trim $t " " -}} {{- $drop := index $tagIgnoreSet (lower $t) -}} {{- if not $drop -}} {{- range $pat := $tagIgnorePatterns -}} {{- if findRE $pat $t -}}{{- $drop = true -}}{{- end -}} {{- end -}} {{- end -}} {{- if and $t (not $drop) -}} {{- $kept = $kept | append $t -}} {{- end -}} {{- end -}} {{- $keywords = delimit $kept ", " -}} {{- end -}} {{- /* Override EXIF with data file values where available */ -}} {{- with $meta.title -}} {{- $title = index . $lang -}} {{- if not $title -}}{{- $title = index . "en" -}}{{- end -}} {{- end -}} {{- /* Fallback: derive a title from the filename when none is set, skipping camera-generated codes like P1002405 / DSC_0753 */ -}} {{- if not $title -}} {{- $stem := strings.TrimSuffix (path.Ext (path.Base $image.Name)) (path.Base $image.Name) -}} {{- $stem = strings.TrimPrefix "archive--" $stem -}} {{- if not (findRE `(?i)^[ _-]?[a-z]{0,4}[ _-]?\d{3,}([ _-][a-z0-9]+)*$` $stem) -}} {{- $title = $stem -}} {{- end -}} {{- end -}} {{- /* DisplayTitle is never empty. $title stays blank for camera-code filenames so the grid does not caption a photo the author never named, but a page heading, a nav link and a related-photo label all need something to print — and the stem IS the name until the file is renamed. Derived once here so those three never disagree. */ -}} {{- $bare := path.Base $image.Name -}} {{- $stemAll := strings.TrimPrefix "archive--" (strings.TrimSuffix (path.Ext $bare) $bare) -}} {{- $displayTitle := $title | default $stemAll -}} {{- /* Alt text: data file only. There used to be an image-alt.html fallback onto the file's own XMP here, which served English to all three languages and let a darktable re-export silently change what the site said; the import moved those 281 values into the YAML and this branch went with them. 60 photos have no alt text yet — that is an editorial backlog, and an empty alt is a warning in the shortcode, not an error. */ -}} {{- $alt := "" -}} {{- with $meta.alt -}} {{- $alt = index . $lang -}} {{- if not $alt -}}{{- $alt = index . "en" -}}{{- end -}} {{- end -}} {{- /* Description: the story, not the alt text. `alt` is the short account of what is visible that a screen reader gets; `description` is the prose the photo page prints. Falls back to alt so a photo with only an alt still reads the way it did before the field existed. */ -}} {{- $description := "" -}} {{- with $meta.description -}} {{- $description = index . $lang -}} {{- if not $description -}}{{- $description = index . "en" -}}{{- end -}} {{- end -}} {{- if not $description -}}{{- $description = $alt -}}{{- end -}} {{- /* License resolution: data file, or a build error. The value is a KEY into data/licenseMap.yaml — the English display strings the entries used to hold were rewritten by the one-time import (docs/concepts/gallery-metadata-import.md), and the aliases that used to resolve them left with it. The Exif.Copyright fallback went too: a licence that lives only inside a JPEG is a licence a re-export can change. Every one of the 648 entries carries a key, so a photo that reaches this without one is new and genuinely unlicensed. */ -}} {{- $licenseValue := "" -}} {{- with $meta.license -}} {{- $licenseValue = . -}} {{- else -}} {{- errorf "No license found for image %s — set `license:` in data/gallery.yaml" (path.Base $image.Name) -}} {{- end -}} {{- $licenseKey := "" -}} {{- $licenseText := "" -}} {{- $licenseUrl := "" -}} {{- $licenseTerms := "" -}} {{- /* Guarded on $licenseValue: errorf does not stop the build, so without it a photo with no licence at all reported this second error too, naming a key of "" and burying the one message that says what to do. */ -}} {{- with $licenseValue -}} {{- with index $licenseIndex (lower .) -}} {{- $licenseKey = .key -}} {{- $licenseText = .label -}} {{- $licenseUrl = .url -}} {{- $licenseTerms = .terms -}} {{- else -}} {{- errorf "License %q for image %s is not a key in data/licenseMap.yaml" $licenseValue (path.Base $image.Name) -}} {{- end -}} {{- end -}} {{- /* Artist resolution: data file > site author.
